sudo mkdir -p /mnt/nas/fotos/destino

Instalar el cliente CIFS
sudo apt update
sudo apt install cifs-utils


Montar a mano para probar, despues ira persostente en fstab
sudo mount -t cifs "//192.168.1.30/piedras/fotos/descarga" /mnt/nas/fotos/destino \
  -o username=usuario,password=oletna2021,uid=$(id -u),gid=$(id -g),vers=3.0

Verificar que funcionó

mount | grep piedras
ls /mnt/nas/fotos/destino
touch /mnt/nas/fotos/destino/test.txt   # confirma que también podés escribir

Desmontar cuando termines la prueba

sudo umount /mnt/nas/fotos/destino

Un tip para cuando pases esto a /etc/fstab (permanente)

No pongas la contraseña en texto plano ahí — usá un archivo de credenciales aparte:
sudo nano /etc/nas-credentials-destino
username=usuario
password=oletna2021


sudo nano /etc/nas-credentials-origen
username=camaras
password=oletna2021


sudo chmod 600 /etc/nas-credentials-destino
sudo chmod 600 /etc/nas-credentials-origen
sudo chown root:root /etc/nas-credentials-destino
sudo chown root:root /etc/nas-credentials-origen

sudo nano /etc/fstab
//192.168.1.30/piedras/fotos/descarga  /mnt/nas/fotos/destino  cifs  credentials=/etc/nas-credentials-destino,uid=1000,gid=1000,_netdev,vers=3.0  0  0
//192.168.1.30/piedras/fotos/camaras  /mnt/nas/fotos/origen  cifs  credentials=/etc/nas-credentials-origen,uid=hach,gid=hach,_netdev,vers=3.0,iocharset=utf8  0  0

Instalo photo_batcher en un entorno virtual venv
Crear el venv e instalar Pillow
sudo mkdir -p /opt/photo_batcher
sudo chown hach:hach /opt/photo_batcher

sudo -u hach python3 -m venv /opt/photo_batcher/venv
sudo -u hach /opt/photo_batcher/venv/bin/pip install --upgrade pip
sudo -u hach /opt/photo_batcher/venv/bin/pip install Pillow

Verificar:
/opt/photo_batcher/venv/bin/python3 -c "import PIL; print(PIL.__version__)"


Creo directorio logs
mkdir -p /opt/photo_batcher/logs

Copiar el script al venv
sudo cp photo_batcher.py /opt/photo_batcher/
sudo chown hach:hach /opt/photo_batcher/photo_batcher.py /opt/photo_batcher/logs

Instalar y arrancar el servicio

sudo cp photo-batcher.service /etc/systemd/system/
sudo systemctl daemon-reload
sudo systemctl enable photo-batcher
sudo systemctl start photo-batcher
sudo systemctl status photo-batcher
